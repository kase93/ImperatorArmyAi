#Independent operations unit objective. Basically used to enable default AI for player as far as possible.

objective_independent = {
	allow = {
		always = yes
	}
	
	interface_priority = 100
	
	#No custom evaluation function.
	mode = none #none/add/replace
	
	#Custom evaluation function. Unit scope. Use scope:province to access the province being evaluated.
	#Negative =&gt; better, positive =&gt; worse. If total sum exceeds 1 million unit will not go there. Beware fixed point overflow at 2 million...
	weight = {
		# ============================================
		# PRIORITY 1: DEFEND THE CAPITAL
		# ============================================
		scope:province = {
			if = {
				limit = {
					# This is our capital
					is_capital = yes
					# There are enemies in the province or adjacent ones
					OR = {
						any_enemy_country = {
							any_army = {
								location = {
									OR = {
										this = ROOT
										is_adjacent_to = ROOT
									}
								}
							}
						}
					}
				}
				# Very negative weight (very attractive) for defending
				weight = -1000000
			}
		}
		
		# ============================================
		# PRIORITY 2: DEFEND NATIONAL TERRITORY
		# ============================================
		scope:province = {
			if = {
				limit = {
					# This is one of our provinces
					owner = {
						country = ROOT
					}
					# Not the capital (already covered above)
					is_capital = no
					# There are enemies in the province or adjacent ones
					OR = {
						any_enemy_country = {
							any_army = {
								location = {
									OR = {
										this = ROOT
										is_adjacent_to = ROOT
									}
								}
							}
						}
					}
				}
				# Very negative weight, but less than the capital
				weight = -500000
			}
		}
		
		# ============================================
		# PRIORITY 3: ATTACK A NEARBY ENEMY
		# ============================================
		scope:province = {
			if = {
				limit = {
					# There are enemy armies in this province
					any_enemy_country = {
						any_army = {
							location = this
						}
					}
					# And it is an enemy province (not ours)
					owner = {
						country = {
							is_enemy_of = ROOT
						}
					}
				}
				# Attack enemy armies on enemy territory
				weight = -300000
			}
		}
		
		# ============================================
		# PRIORITY 4: SIEGE NEARBY TERRITORY
		# ============================================
		scope:province = {
			if = {
				limit = {
					# It is an enemy province
					owner = {
						country = {
							is_enemy_of = ROOT
						}
					}
					# No enemy army (already covered in priority 3)
					NOT = {
						any_enemy_country = {
							any_army = {
								location = this
							}
						}
					}
					# It is adjacent to our territory
					any_neighbor_province = {
						owner = {
							country = ROOT
						}
					}
				}
				# Siege nearby enemy territory
				weight = -200000
			}
		}
		
		# ============================================
		# PRIORITY 5: SIEGE THE ENEMY CAPITAL
		# ============================================
		scope:province = {
			if = {
				limit = {
					# It is the enemy capital
					is_capital = yes
					owner = {
						country = {
							is_enemy_of = ROOT
						}
					}
					# No enemy army (already covered in priority 3)
					NOT = {
						any_enemy_country = {
							any_army = {
								location = this
							}
						}
					}
				}
				# Siege the enemy capital (high priority to win the war)
				weight = -250000
			}
		}
		
		# ============================================
		# PRIORITY 6: ATTACK RESOURCE-RICH PROVINCES
		# ============================================
		scope:province = {
			if = {
				limit = {
					# It is an enemy province
					owner = {
						country = {
							is_enemy_of = ROOT
						}
					}
					# Tiene recursos valiosos
					OR = {
						has_trade_goods = {
							trade_goods = gold
						}
						has_trade_goods = {
							trade_goods = iron
						}
						has_trade_goods = {
							trade_goods = grain
						}
					}
					# Not adjacent (already covered in priority 4)
					NOT = {
						any_neighbor_province = {
							owner = {
								country = ROOT
							}
						}
					}
				}
				# Attack provinces with resources
				weight = -150000
			}
		}
		
		# ============================================
		# PRIORITY 7: SIEGE DISTANT PROVINCES
		# ============================================
		scope:province = {
			if = {
				limit = {
					# It is an enemy province
					owner = {
						country = {
							is_enemy_of = ROOT
						}
					}
					# No enemy army
					NOT = {
						any_enemy_country = {
							any_army = {
								location = this
							}
						}
					}
					# Not adjacent (already covered in priority 4)
					NOT = {
						any_neighbor_province = {
							owner = {
								country = ROOT
							}
						}
					}
				}
				# Siege distant enemy provinces
				weight = -100000
			}
		}
		
		# ============================================
		# PRIORITY 8: REINFORCE BORDERS
		# ============================================
		scope:province = {
			if = {
				limit = {
					# This is one of our provinces
					owner = {
						country = ROOT
					}
					# It is on the border (adjacent to an enemy)
					any_neighbor_province = {
						owner = {
							country = {
								is_enemy_of = ROOT
							}
						}
					}
					# No enemies nearby (already covered in priority 2)
					NOT = {
						any_enemy_country = {
							any_army = {
								location = {
									OR = {
										this = ROOT
										is_adjacent_to = ROOT
									}
								}
							}
						}
					}
				}
				# Reinforce borders to prevent invasions
				weight = -50000
			}
		}
		
			# ============================================
		# PRIORITY 9: GARRISON INTERIOR PROVINCES
		# ============================================
		scope:province = {
			if = {
				limit = {
					# This is one of our provinces
					owner = {
						country = ROOT
					}
					# It is not on the border
					NOT = {
						any_neighbor_province = {
							owner = {
								country = {
									is_enemy_of = ROOT
								}
							}
						}
					}
					# Not the capital (already covered in priority 1)
					is_capital = no
				}
				# Light garrison in interior provinces
				weight = -10000
			}
		}
		
		# ============================================
		# PRIORITY 10: REINFORCE THE CAPITAL (when safe)
		# ============================================
		scope:province = {
			if = {
				limit = {
					# This is our capital
					is_capital = yes
					# No enemies nearby (already covered in priority 1)
					NOT = {
						any_enemy_country = {
							any_army = {
								location = {
									OR = {
										this = ROOT
										is_adjacent_to = ROOT
									}
								}
							}
						}
					}
				}
				# Keep a garrison in the capital even when it is safe
				weight = -20000
			}
		}
		
		# ============================================
		# PRIORITY 11: NEARBY NEUTRAL PROVINCES
		# ============================================
		scope:province = {
			if = {
				limit = {
					# It is a neutral province (neither ours nor enemy)
					owner = {
						country = {
							NOT = {
								OR = {
									this = ROOT
									is_enemy_of = ROOT
								}
							}
						}
					}
					# It is adjacent to our territory
					any_neighbor_province = {
						owner = {
							country = ROOT
						}
					}
				}
				# Scout/secure nearby neutral provinces
				weight = -5000
			}
		}
		
		# ============================================
		# PRIORITY 12: NO PREFERRED TARGET
		# ============================================
		# If no condition holds, the weight is 0 (neutral)
		# The unit will stay where it is or look for another objective
	}
}
